%
% Defines \ClebschSeriesMap{<node>}, where <node> is one of
% I, II, III, IV, V, or none.  The named node is drawn with a heavier rule
% and a light grey tint ("you are here"); every other node is plain.
%
% Faithful to the ASCII master in
% notes/2026-08-03-clebsch-program-unity-review.md.  Two conventions are
% load-bearing and must survive any edit:
%   * the Paper IV edge is DASHED, because no theorem carries its
%     reconstructed plane to the oriented cubic torsor;
%   * the Paper V node states an intent, not a result.

\providecommand{\ClebschSeriesMap}[1]{%
\csfSetHighlight{#1}%
\begin{tikzpicture}[
  every node/.style={outer sep=0pt},
]

\node[\csfHL{III},text width=64mm] (pIII) at (0,4.7)
  {\textbf{Paper III} --- passages\\[1pt]
   arithmetic source: \(z^{2}=5J_{0}\)\\
   operator shadows: \(\operatorname{Pf}\), \(\det\),
   \(\textstyle\bigwedge^{3}\), polar};

\node[csfcentre,text width=37mm] (z) at (0,1.85)
  {\textbf{GOLDEN TORSOR}\\[1pt]
   \(B^{2}=5I\)\\
   \(c_{ijk}=B_{ij}B_{jk}B_{ki}\)\\
   one oriented cubic \(Z\)};

\node[\csfHL{I},text width=37mm] (pI) at (-5.45,1.85)
  {\textbf{Paper I}\\[1pt]
   deep-hole syndrome and decoder data recover \(B\) and its support
