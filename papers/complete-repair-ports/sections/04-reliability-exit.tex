\section{Consequences of the threshold}
\label{sec:consequences}

The fixed-target relative weights sit between ordinary generalized weights
and worst-case cooperative locality.  Let $d_e(C^\perp)$ denote the $e$th
generalized Hamming weight of $C^\perp$.  For an $e$-set $P\subseteq E$, let
$\kappa_C(P)$ be the least $|H|$ for which some $e$-dimensional subcode
$A\leq C^\perp$, supported on $P\cup H$, restricts isomorphically to
$\F_q^P$.  Set $\kappa_C(P)=\infty$ when no such subcode exists.  Thus
$\kappa_C(P)$ is the minimum number of helpers needed to recover all
coordinates in $P$ simultaneously.

\begin{theorem}[Best-target generalized weight and first escape cost]
\label{thm:best-target-ghw}
\coverage{absent}%
For $1\leq e\leq\dim C^\perp$,
\begin{equation}
 \min_{\substack{P\subseteq E\\|P|=e}}\kappa_C(P)
 =d_e(C^\perp)-e.
 \label{eq:best-target-ghw}
\end{equation}
Consequently, if every $e$-set is recoverable and
$r_e^{\mathrm{coop}}(C)=\max_{|P|=e}\kappa_C(P)$, then
\[
 r_e^{\mathrm{coop}}(C)\geq d_e(C^\perp)-e.
\]
For a fixed inner code in the eventual concatenation regime of
Theorem~\ref{thm:objectwise-confinement}, the least first nonconfinement cost
over all $e$-coordinate target sets is
\begin{equation}
 d_e(C^\perp)-e+d(C^\perp).
 \label{eq:best-target-escape}
\end{equation}
\end{theorem}

\begin{proof}
A normalized identity system on $P$ spans an $e$-dimensional subcode of
$C^\perp$ whose support is $P$ together with its helper union.  Hence
$d_e(C^\perp)\leq e+\kappa_C(P)$.

Conversely, let $A\leq C^\perp$ have dimension $e$ and support size
$d_e(C^\perp)$.  An information set $P$ of a generator matrix of $A$ has
size $e$, and restriction $A\to\F_q^P$ is an isomorphism.  Row reduction on
these columns gives normalized identity equations whose helper union is
$\supp(A)\setminus P$.  This proves~\eqref{eq:best-target-ghw}; the
cooperative-locality inequality follows by taking the maximum instead of the
minimum.

It remains to justify the escape formula when the target columns indexed by
$P$ are dependent.  Put $U=\operatorname{im}G_P$.  A normalized identity
system on $\F_q^P$ and a normalized recovery system on $U$ have the same
minimum helper union.  Indeed, restricting an identity system along a linear
section $\alpha:U\to\F_q^P$ gives a system on $U$.  Conversely, given a
system on $U$, write
\[
 a=\alpha(G_Pa)+\bigl(a-\alpha(G_Pa)\bigr)
 \qquad(a\in\F_q^P).
\]
The second summand lies in $\ker G_P$ and hence is a target-only dual
equation, so adjoining these kernel equations recovers the full identity
without adding helpers.  Thus the common minimum is $\kappa_C(P)$.

Now repeat the block-functional argument of
Theorem~\ref{thm:objectwise-confinement}.  Once the outer-functional case is
excluded, the target block of any identity system uses at least
$\kappa_C(P)$ helpers.  If the system is nonconfined, a nonzero equation map
in another block has union support at least $d(C^\perp)$; the two supports
are disjoint.  Conversely, attach a minimum-weight word of $C^\perp$ in a
second block through any nonzero linear functional on $\F_q^P$.  Hence the
first nonconfinement cost for this $P$ is
$\kappa_C(P)+d(C^\perp)$.  Minimizing over $P$ and using
\eqref{eq:best-target-ghw} proves~\eqref{eq:best-target-escape}.
\uses{thm:objectwise-confinement}%
\proves{thm:best-target-ghw}%
\end{proof}

Put $k=\dim I$ and $b=\rank G_J$.  The exact sequence and the relative and
ordinary Singleton bounds give
\begin{equation}
 M_t(D_P,K_P)\leq b-\ell+t,
 \qquad d(I^\perp)\leq k+1.
 \label{eq:singleton-bounds}
\end{equation}

\begin{theorem}[MDS thresholds and rigidity]
\label{thm:mds-thresholds}
\coverage{absent}%
For $1\leq t\leq\ell$,
\begin{equation}
 M_t(D_P,K_P)+d(I^\perp)
 \leq2k-\ell+t+1.
 \label{eq:threshold-ceiling}
\end{equation}
Suppose $I$ is MDS, and put
\[
 u=\min\{k,|P|\},\qquad b=\min\{k,|J|\}.
\]
Then $\ell=u+b-k$.  If $\ell\geq1$, then, for $1\leq t\leq\ell$,
\begin{equation}
 M_t(D_P,K_P)=k-u+t,
 \qquad
 M_t(D_P,K_P)+d(I^\perp)=2k-u+t+1.
 \label{eq:mds-staircase}
\end{equation}
If $|J|\geq k$, then $\ell=u$ and these values attain the ceiling in
\eqref{eq:threshold-ceiling} at every rank.
Conversely, when $\ell\geq1$, equality in~\eqref{eq:threshold-ceiling} at
$t=1$ forces $I$ to be MDS and forces equality at every rank.
\end{theorem}

\begin{proof}
The ceiling follows from~\eqref{eq:singleton-bounds} and $b\leq k$.  For an
MDS code the column matroid is uniform, so $\rank G_P=u$, $\rank G_J=b$,
$\ell=u+b-k$, and, for $H\subseteq J$ with $|H|=h\leq k$,
\[
 \dim\bigl(\operatorname{span}G_P\cap\operatorname{span}G_H\bigr)
 =\max\{0,u+h-k\}.
\]
Thus Proposition~\ref{prop:relative-profile} gives $M_t=k-u+t$ for
$1\leq t\leq\ell$; here $k-u+t\leq b\leq|J|$.  Now
$d(I^\perp)=k+1$, and if $|J|\geq k$, then $b=k$ and $\ell=u$, proving the
remaining direct assertions.

Equality at $t=1$ forces equality in both Singleton bounds and $b\leq k$.
Hence $d(I^\perp)=k+1$, so $I$ is MDS, and $M_1=b-\ell+1$.  Strict growth
then gives $M_t\geq b-\ell+t$, while relative Singleton gives the reverse
inequality; every rank is extremal.
\uses{prop:relative-profile}%
\proves{thm:mds-thresholds}%
\end{proof}

We next record the global parameters and the density of the transferred
data.  Let the inner code have parameters $[m,k,d]_q$, so
$[L:\F_q]=k$.  Let $K_N=\dim_L O_N$ and $D_N=d(O_N)$.

\begin{corollary}[Positive-density realization]
\label{cor:positive-density}
\coverage{absent}%
Assume $d(O_N^\perp)\to\infty$.  Below any fixed threshold supplied by
Theorem~\ref{thm:objectwise-confinement} or
Theorem~\ref{thm:ranked-confinement}, for all sufficiently large $N$ every
inner normalized recovery system and exact helper support is copied in each
of the $N$ blocks.  Each fixed inner coordinate type has density $1/m$ among
the concatenated coordinates; the union of the types in $P$ has density
$p/m$.

The concatenated code has length $mN$, dimension $kK_N$, and minimum distance
at least $dD_N$.  If
\[
 \frac{K_N}{N}\longrightarrow R>0,
 \qquad
 \liminf_N\frac{D_N}{N}\geq\delta>0,
\]
then the concatenated family has rate $kR/m$ and relative distance at least
$d\delta/m$.  Outer families satisfying all three hypotheses exist over
every finite field $L$.
\end{corollary}

\begin{proof}
The transfer statement is the zero-extension bijection in the confinement
theorems, applied independently in every block.  There are $mN$ coordinates,
and each inner coordinate occurs once in each block, which gives the density
claims.  The blockwise inner encoder is injective, so restriction of scalars
gives dimension $kK_N$.  A nonzero outer word has at least $D_N$ nonzero
symbols, each encoding to an inner word of weight at least $d$.  This proves
the distance and asymptotic assertions.

For the existence statement, write $Q=|L|$.  Choose $0<R<1$ and positive
$\delta,\delta^\perp$ such that
\[
 H_Q(\delta)<1-R,
 \qquad
 H_Q(\delta^\perp)<R,
\]
where $H_Q$ is the $Q$-ary entropy function.  The first-moment argument for a
random $K_N$-dimensional $L$-linear code bounds the expected numbers of
nonzero primal and dual words below the two cutoffs, up to subexponential
factors, by
\[
 Q^{N(H_Q(\delta)+R-1)}
 \qquad\text{and}\qquad
 Q^{N(H_Q(\delta^\perp)-R)},
\]
respectively.  Both tend to zero by the displayed inequalities, so one may
choose a sequence with rate tending to $R$, primal relative distance at least
$\delta$, and dual relative distance at least $\delta^\perp$; see also
\cite[Chapter~17]{MacWilliamsSloane1977}.
\uses{thm:objectwise-confinement, thm:ranked-confinement}%
\proves{cor:positive-density}%
\end{proof}

Exact support transfer also preserves the fractional load-balancing region
used in service-rate problems.  We state the consequence because it requires
only the inclusion-minimal recovery sets, whereas the theorem transfers all
normalized coefficients as well.

Fix one inner target block with helper set $J$.  Let $\mathcal A$ be a finite
set of demands on that block, with demand $a$ represented by a nonzero target
subspace $T_a\leq W_P$, and let
$\mathcal M_a^{(\leq r)}$ be the inclusion-minimal exact helper supports of
size at most $r$ for $a\in\mathcal A$.  For a nonnegative helper-capacity
vector $c$, define
\[
 \Lambda_{\mathcal A}^{(\leq r)}(c)=
 \left\{\lambda\geq0:\begin{array}{l}
 \text{there are }f_{a,R}\geq0\text{ with }
 \lambda_a=\sum_{R\in\mathcal M_a^{(\leq r)}}f_{a,R},\\[2pt]
 \displaystyle\sum_a\sum_{R\ni j}f_{a,R}\leq c_j
 \text{ for every helper }j
 \end{array}\right\}.
\]
This is the bounded service-rate region of the recovery-set system
\cite{Mazumdar2023,ChoudharyEtAl2026}.

\begin{corollary}[Transfer of bounded service-rate regions]
\label{cor:service-rate-transfer}
\coverage{absent}%
Let $O\leq L^N$ be an outer code with $N\geq2$ whose projection onto the
target block is nonzero.  Suppose
\[
 r<\Gamma_{j,T_a}(O,I)
 \qquad(a\in\mathcal A).
\]
Transport capacities from the inner helper set to its copy in that block of
$O\circ I$.  Then the inner and concatenated bounded service-rate regions are
equal.  Capacities assigned outside the target block do not change this
equality.  In particular, the hypothesis follows from
$d(O^\perp)>r+1$ and the corresponding inner inequalities.  Consequently,
for an outer family with $d(O_N^\perp)\to\infty$, the equality holds for all
sufficiently large $N$ whenever those inner inequalities hold.
\end{corollary}

\begin{proof}
Theorem~\ref{thm:ungated-ranked-confinement} confines every cost-$\leq r$
system for every demand.  Thus restriction and zero-extension copy every
inclusion-minimal support and create no cross-block minimal support.  The
helper bijection therefore transports the
feasible variables $f_{a,R}$ and every capacity inequality in both
directions.  Allowing the full upward-closed recovery-set family gives the
same region: flow assigned to a nonminimal set can be moved to a contained
minimal set and weakly decreases every helper load.  Thus cross-block
supersets and capacities outside the target block are irrelevant.
\uses{thm:ungated-ranked-confinement, thm:objectwise-confinement}%
\proves{cor:service-rate-transfer}%
\end{proof}
